cmake_minimum_required(VERSION 3.24)
project(rmcs_yaw_identification)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rmcs_executor REQUIRED)
find_package(rmcs_description REQUIRED)
find_package(fast_tf REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/controller.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME} rclcpp pluginlib rmcs_executor rmcs_description fast_tf std_msgs std_srvs)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(rmcs_executor plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(engine_test test/engine_test.cpp)
  target_include_directories(engine_test PRIVATE include)
  ament_target_dependencies(engine_test rclcpp)
  add_test(NAME engine_test COMMAND engine_test)
  add_executable(plugin_check test/plugin_check.cpp)
  ament_target_dependencies(plugin_check rclcpp pluginlib rmcs_executor)
  add_test(NAME plugin_check COMMAND plugin_check)
endif()
ament_package()
